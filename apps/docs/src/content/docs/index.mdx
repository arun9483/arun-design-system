import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Tokens, not themes">
    Every colour, space and radius is a CSS custom property. No theme object, no runtime style
    engine, no CSS-in-JS. `@arun-dev/tokens` ships zero JavaScript.
  </Card>
  <Card title="Brands from one colour">
    `createBrand({seed})` generates a full palette and semantic layer at build time. Switch the
    brand selector in the header — no component re-renders differently.
  </Card>
  <Card title="Render anything">
    Every component takes a `render` prop, so a chip can be an `li`, a button can be a router link,
    and a card can be a labelled `nav`.
  </Card>
  <Card title="Behaviour without styling">
    `@arun-dev/headless` ships the behaviour, keyboard handling and accessibility with no CSS and no
    class names. State reaches your stylesheet as `data-*` attributes.
  </Card>
  <Card title="Owns its contract">
    A build-time test fails if the library reads a custom property the token package does not
    define. No silently-unstyled components.
  </Card>
</CardGrid>
